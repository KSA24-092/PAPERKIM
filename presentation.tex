    \begin{itemize}
      \item \textbf{Bridges Conference}: 수학과 예술의 교차점을 다루는 국제 학회
      \item Computational String Art에 대한 \textbf{탐욕 알고리즘(greedy algorithm)} 제안

  \vspace{0.2cm}
  \textbf{논문의 핵심 기여}:
  \begin{enumerate}
    \item \textbf{선 선택 휴리스틱}: 오차 최소화 vs.\ 최대 암도 선택의 비교
    \item \textbf{정지 조건}: 새로운 선이 오차를 줄이지 못하면 자동 종료

\begin{frame}{String Art 문제의 수학적 정의}
  \begin{columns}[T]

      \textbf{입력}:
      \begin{itemize}
        \item 그레이스케일 이미지 $I_{\text{target}}$ ($M \times M$ 픽셀)
        \item 원 위에 등간격 배치된 $N$개 핀
      \end{itemize}

      \vspace{0.2cm}
      \textbf{출력}:
      \begin{itemize}
        \item 핀 쌍의 순서 집합 (실의 연결 순서)
        \item 각 실의 밝기(또는 불투명도)
      \end{itemize}

      \vspace{0.2cm}
      \textbf{핀 좌표}:
      \[
        P_k = \left(\tfrac{M}{2} + r\cos\tfrac{2\pi k}{N},\;
              \tfrac{M}{2} + r\sin\tfrac{2\pi k}{N}\right)
      \]

      \vspace{0.2cm}
      \textbf{목표}: 실 이미지와 원본 사이의 \textbf{오차 최소화}
      \[
        \min \sum_{r,c} \bigl(R(r,c) - I_{\text{target}}(r,c)\bigr)^2
      \]

\section{탐욕 알고리즘의 수학}
% =================================================================

\begin{frame}{Bridges 논문의 탐욕 알고리즘}
  \begin{block}{기본 아이디어 (3단계 반복)}
    \begin{enumerate}
      \item 가능한 모든 선분 중 \textbf{오차를 가장 많이 줄이는 선분} 선택
      \item 선분을 캔버스에 \textbf{그리기} (이미지에 반영)
      \item 오차가 줄지 않으면 \textbf{종료}
    \end{enumerate}
  \end{block}

  \vspace{0.3cm}
  \textbf{선 선택 점수} — 각 후보 선분 $\ell$에 대해:
  \[
    \text{score}(\ell) = \sum_{i \in \ell} \bigl(\text{현재 오차}_i^2 - \text{선 추가 후 오차}_i^2\bigr)
  \]

  \vspace{0.2cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{불투명(opaque) 실}
        선 픽셀을 검정(0)으로 덮어쓰기\\
        $R'_i = 0$
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{반투명(non-opaque) 실}
        기존 밝기를 비율로 감소\\
        $R'_i = R_i \cdot (1 - \alpha)$
      \end{exampleblock}
    \end{column}
  \end{columns}

  \vspace{0.3cm}
  $N$개 핀 → $\frac{N(N{-}1)}{2}$개 후보 선분 → 매 반복마다 전부 평가
\end{frame}

\begin{frame}{투과율 모델과 에너지 함수}
  \textbf{반투명 실}의 물리적 모델 (Beer--Lambert 법칙):

  \vspace{0.2cm}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{align}
        T(r,c) &\in [0,1] \quad \text{(투과율)} \\[2pt]
        R(r,c) &= 255 \cdot T(r,c) \quad \text{(렌더 밝기)} \\[4pt]
        \text{선 추가:}\quad T'_i &= T_i \cdot (1 - \alpha \cdot v_i)
      \end{align}

      \begin{itemize}
        \item $\alpha$: 실 불투명도 ($0 < \alpha \le 1$)
        \item $v_i$: 해당 픽셀의 안티앨리어싱 가중치
      \end{itemize}

      \vspace{0.2cm}
      $\Rightarrow$ 실이 겹칠수록 효과가 \textbf{자연히 포화}\\
      \quad\; (가산 모델의 과도 암화 문제 해결)
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{에너지 함수}:
      \[
        E = \sum_{r,c} \bigl(R(r,c) - I_{\text{target}}\bigr)^2
      \]

      \textbf{한 가닥 추가 시 에너지 감소량}:
      \begin{align*}
        s_i &= R_i \cdot \alpha v_i \\
        d_i &= R_i - I_i \\[4pt]
        \Delta E &= \sum_i s_i(2d_i - s_i)
      \end{align*}

      $\Delta E > 0$ → 좋은 선택\\
      $\Delta E \le 0$ → 정지
    \end{column}
  \end{columns}
\end{frame}

\section{우리의 개선점 (Bridges 논문과의 차별점)}
% =================================================================

\begin{frame}{Bridges 논문의 한계점}
  Demoussel et al.\ (2022) 논문은 실용적이지만 다음과 같은 한계가 있다:

  \vspace{0.3cm}
  \begin{enumerate}
    \item \textbf{단순 탐욕}: 매 단계 1-step만 고려 → 근시안적(myopic) 선택
    \item \textbf{되돌림 불가}: 한 번 그린 선은 제거 불가
          (Birsak et al.은 제거 가능하나 2시간 이상 소요)
    \item \textbf{균일 가중치}: 모든 픽셀을 동등하게 취급
          (중요도 맵은 있으나 에지 기반 자동화 부족)
    \item \textbf{정규화 부재}: 실 수 제어가 고정 상한 또는 단순 정지 조건에 의존
    \item \textbf{후처리 없음}: 탐욕 결과가 곧 최종 결과
  \end{enumerate}

  \vspace{0.3cm}
  \begin{alertblock}{핵심 질문}
    탐욕 알고리즘의 빠른 속도를 유지하면서,
    결과 품질을 \textbf{어떻게 더 끌어올릴 수 있을까?}
  \end{alertblock}
\end{frame}

\begin{frame}{우리의 개선: 수학적 확장}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. 가중 에너지 함수}
        에지 검출 기반 중요도 맵 $W$:
        \[
          W = (1 + \lambda \cdot E_{\text{edge}}) \cdot M_{\text{circle}}
        \]
        → 윤곽선에 높은 가중치 부여\\
        → \textbf{구조적 디테일 우선 복원}
      \end{block}

      \vspace{0.2cm}
      \begin{block}{2. 희소 정규화}
        \[
          \Delta E_{\text{reg}} = \Delta E - C_s - C_r \!\cdot\! u - C_n \!\cdot\! \mathbb{1}_{\text{new}}
        \]
        → 실 수, 핀쌍 재사용, 새 핀 도입에 페널티\\
        → \textbf{불필요한 실 억제}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{3. 2-step Beam Lookahead}
        상위 $K$개 후보에 대해 1-step 더 예측:
        \[
          \text{score} = \Delta E_{\text{reg}} + w \cdot \Delta E_{\text{next}}
        \]
        → 근시안적 선택 문제 완화
      \end{block}

      \vspace{0.2cm}
      \begin{block}{4. 다단계 후처리}
        \begin{itemize}
          \item \textbf{정련(Refinement)}: 시퀀스 재생성,\\
                각 선택을 재평가하여 교체
          \item \textbf{압축}: $(a{\to}b{\to}c)$를 $(a{\to}c)$로 축약
          \item \textbf{시뮬레이티드 어닐링}: 전역 탐색으로 탈출
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{차별점 요약: Bridges vs.\ 우리의 구현}
  \begin{center}
  \small
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular}{p{3.2cm}p{4.8cm}p{5.2cm}}
    \toprule
    항목 & \textbf{Bridges 2022} & \textbf{우리의 구현 (개선)} \\
    \midrule
    탐색 전략 & 1-step 탐욕 & 2-step Beam Lookahead \\
    에너지 함수 & 균일 가중 RMS & 에지 기반 중요도 가중 $L^2$ \\
    정규화 & 없음 (고정 상한) & 희소 정규화 ($C_s, C_r, C_n$) \\
    후처리 & 없음 & 정련 + 압축 + 시뮬레이티드 어닐링 \\
    되돌림 & 불가 & 정련 단계에서 사실상 재선택 \\
    물리 모델 & 가산/반투명 혼용 & Beer--Lambert 곱셈 모델 일관 적용 \\
    실행 시간 목표 & $< 1$분 & 수 분 (품질 우선, 병렬화로 보상) \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.3cm}
  \begin{alertblock}{요약}
    Bridges 논문의 \textbf{빠르고 단순한 탐욕 프레임워크}를 기반으로,
    \textbf{물리적 정확성 + 다단계 최적화 + 정규화}를 결합하여
    품질을 끌어올리는 것이 핵심 차별점
  \end{alertblock}
\end{frame}

\section{수학적 가치}
% =================================================================

\begin{frame}{수학적 가치}
  \begin{block}{1. 이산 최적화 (Discrete Optimization)}
    \begin{itemize}
      \item $N$개 핀, 최대 $T$개 실 → 해 공간 크기 $\sim N^T$
      \item NP-hard 조합 문제를 \textbf{탐욕 + Beam 탐색}으로 근사
      \item 여행 세일즈맨 문제(TSP)와 유사한 경로 최적화 구조
    \end{itemize}
  \end{block}

  \begin{block}{2. 물리 기반 수학 모델링}
    \begin{itemize}
      \item Beer--Lambert 법칙: 빛 투과의 지수적 감쇄를 곱셈 모델로 표현
      \item 에너지 감소량의 \textbf{폐형식(closed-form) 유도}:
            $\Delta E = \sum W_i \cdot s_i(2d_i - s_i)$
      \item 연속적 물리 현상을 이산 알고리즘에 연결
    \end{itemize}
  \end{block}

  \begin{block}{3. 정규화 이론}
    \begin{itemize}
      \item $L_0/L_1$ 희소 정규화: 모델 복잡도와 충실도의 균형
      \item 통계학/기계학습의 핵심 개념을 예술 생성에 적용
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{예술적 가치}
